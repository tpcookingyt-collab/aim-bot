-- Executor detection fingerprint

    -- Check known executor globals
    if syn then return "Synapse X"
    elseif KRNL_LOADED then return "Krnl"
    elseif Delta then return "Delta"
    elseif getgenv().is_delta_executor then return "Delta"
    elseif fluxus then return "Fluxus"
    elseif Celery then return "Celery"
    elseif Electron then return "Electron"
    elseif getgenv().MACSPLOIT_EXECUTOR then return "MacSploit"
    elseif getgenv().NIHON then return "Nihon"
    elseif getgenv().Vega then return "Vega X"
    elseif getgenv().ARCEUS_X then return "Arceus X"
    elseif getgenv().JJSploit then return "JJSploit"
    elseif getgenv().Oxygen then return "Oxygen U"
    elseif getgenv().COCO_Z then return "Comet"
    elseif getgenv().SCRIPT_HUB then return "Script-Ware"

-- Check if already has other cheats loaded (foreign globals)
local function GetCheatSignals()
    local signals = {}
    local known = {
        "esp", "aimbot", "fly", "noclip", "speedhack",
        "wallhack", "triggerbot", "bunnyhop", "spinbot",
        "fovchanger", "bhop", "killaura",
    }
    local env = getgenv and getgenv() or {}
    for _, k in ipairs(known) do
        if env[k] ~= nil then
            table.insert(signals, k)
        end
    end
    return #signals > 0 and table.concat(signals, ", ") or "None detected"
end

local function HasWall(origin, target)
    if not Settings.WallCheck then return false end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local chars = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then table.insert(chars, p.Character) end
    end
    params.FilterDescendantsInstances = chars
    local result = workspace:Raycast(origin, (target - origin).Unit * (target - origin).Magnitude, params)
    return result ~= nil
end

local function WorldToScreen(pos)
    local screen, onScreen = Camera:WorldToViewportPoint(pos)
    return Vector2.new(screen.X, screen.Y), onScreen, screen.Z
end

local function GetCenter()
    return Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
end

local function GetFOVDist(worldPos)
    local screen, onScreen = WorldToScreen(worldPos)
    if not onScreen then return math.huge end
    return (screen - GetCenter()).Magnitude
end

local function GetBoundingBox(character)
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local head   = character:FindFirstChild("Head")
    local topPos = head and head.Position or (root.Position + Vector3.new(0,2.5,0))
    local botPos = root.Position - Vector3.new(0,3,0)
    local topScreen, topOn  = WorldToScreen(topPos)
    local botScreen, botOn  = WorldToScreen(botPos)
    if not topOn and not botOn then return nil end
    local h = math.abs(topScreen.Y - botScreen.Y) * Settings.ESPBoxScale
    local w = h * 0.5
    return {
        X      = botScreen.X - w/2,
        Y      = topScreen.Y,
        Width  = w,
        Height = h,
        Center = Vector2.new(botScreen.X, (topScreen.Y+botScreen.Y)/2),
        TopCenter    = topScreen,
        BottomCenter = botScreen,
    }
end

-- Distance fade alpha
local function FadeAlpha(dist, maxDist, baseAlpha)
    if not Settings.DistanceFade then return 1 - baseAlpha end
    local fade = math.clamp(1 - (dist / maxDist), 0.1, 1)
    return fade * (1 - baseAlpha)
end

-- ══════════════════════════════════════════
-- 3D BOX CORNERS (8 vertices of bounding box)
-- ══════════════════════════════════════════
local function Get3DBoxEdges(character)
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local head   = character:FindFirstChild("Head")
    local topY   = head and head.Position.Y + 0.5 or root.Position.Y + 3
    local botY   = root.Position.Y - 3
    local cx, cz = root.Position.X, root.Position.Z
    local hw = 1.2 -- half-width

    local corners3D = {
        Vector3.new(cx-hw, topY, cz-hw),
        Vector3.new(cx+hw, topY, cz-hw),
        Vector3.new(cx+hw, topY, cz+hw),
        Vector3.new(cx-hw, topY, cz+hw),
        Vector3.new(cx-hw, botY, cz-hw),
        Vector3.new(cx+hw, botY, cz-hw),
        Vector3.new(cx+hw, botY, cz+hw),
        Vector3.new(cx-hw, botY, cz+hw),
    }

    local edges = {
        {1,2},{2,3},{3,4},{4,1}, -- top face
        {5,6},{6,7},{7,8},{8,5}, -- bottom face
        {1,5},{2,6},{3,7},{4,8}, -- verticals
    }

    local screenCorners = {}
    local allOn = true
    for i, v in ipairs(corners3D) do
        local s, on = WorldToScreen(v)
        if not on then allOn = false end
        screenCorners[i] = {screen=s, on=on}
    end

    return screenCorners, edges, allOn
end

-- DRAW CORNER BOX
-- ══════════════════════════════════════════
local function DrawCornerBox(lines, box, color, alpha, thick)
    local x,y,w,h = box.X, box.Y, box.Width, box.Height
    local cw, ch  = w*0.25, h*0.25
    local segs = {
        {Vector2.new(x,y),     Vector2.new(x+cw,y)},
        {Vector2.new(x,y),     Vector2.new(x,y+ch)},
        {Vector2.new(x+w,y),   Vector2.new(x+w-cw,y)},
        {Vector2.new(x+w,y),   Vector2.new(x+w,y+ch)},
        {Vector2.new(x,y+h),   Vector2.new(x+cw,y+h)},
        {Vector2.new(x,y+h),   Vector2.new(x,y+h-ch)},
        {Vector2.new(x+w,y+h), Vector2.new(x+w-cw,y+h)},
        {Vector2.new(x+w,y+h), Vector2.new(x+w,y+h-ch)},
    }
    for i,s in ipairs(segs) do
        lines[i].From=s[1]; lines[i].To=s[2]
        lines[i].Color=color; lines[i].Transparency=alpha
        lines[i].Thickness=thick; lines[i].Visible=true
    end
end

-- ══════════════════════════════════════════
-- UPDATE SINGLE ESP
-- ══════════════════════════════════════════
local function UpdateESP(player, data)
    local char = player.Character
    if not char then
        HideLines(data.CornerBox); HideLines(data.Box3D)
        HideLines(data.Skeleton)
        data.NameLabel.Visible=false; data.HealthBG.Visible=false
        data.HealthFill.Visible=false; data.DistLabel.Visible=false
        data.Tracer.Visible=false; data.Arrow.Visible=false
        data.WeaponLabel.Visible=false; data.Highlight.Enabled=false
        return
    end

    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health<=0 then
        HideLines(data.CornerBox); HideLines(data.Box3D)
        HideLines(data.Skeleton)
        data.NameLabel.Visible=false; data.HealthBG.Visible=false
        data.HealthFill.Visible=false; data.DistLabel.Visible=false
        data.Tracer.Visible=false; data.Arrow.Visible=false
        data.WeaponLabel.Visible=false; data.Highlight.Enabled=false
        return
    end

    local camPos = Camera.CFrame.Position
    local dist   = (root.Position - camPos).Magnitude

    if dist > Settings.ESPMaxDistance then
        HideLines(data.CornerBox); HideLines(data.Box3D)
        HideLines(data.Skeleton)
        data.NameLabel.Visible=false; data.HealthBG.Visible=false
        data.HealthFill.Visible=false; data.DistLabel.Visible=false
        data.Tracer.Visible=false; data.Arrow.Visible=false
        data.WeaponLabel.Visible=false; data.Highlight.Enabled=false
        return
    end

    local color = GetESPColor(player)
    local alpha = FadeAlpha(dist, Settings.ESPMaxDistance, Settings.ESPTransparency)

    -- 2D Corner Box
    if Settings.Box2D then
        local box = GetBoundingBox(char)
        if box then
            DrawCornerBox(data.CornerBox, box, color, alpha, Settings.ESPOutlineThick)
        else
            HideLines(data.CornerBox)
        end
    else
        HideLines(data.CornerBox)
    end

    -- 3D Box
    if Settings.Box3D then
        local corners, edges = Get3DBoxEdges(char)
        if corners then
            for i, edge in ipairs(edges) do
                local a = corners[edge[1]]
                local b = corners[edge[2]]
                local line = data.Box3D[i]
                if a.on and b.on then
                    line.From=a.screen; line.To=b.screen
                    line.Color=color; line.Transparency=alpha
                    line.Thickness=Settings.ESPOutlineThick
                    line.Visible=true
                else
                    line.Visible=false
                end
            end
        else
            HideLines(data.Box3D)
        end
    else
        HideLines(data.Box3D)
    end

    -- Skeleton
    if Settings.SkeletonESP then
        for i, pair in ipairs(BONES) do
            local pA = char:FindFirstChild(pair[1])
            local pB = char:FindFirstChild(pair[2])
            local line = data.Skeleton[i]
            if pA and pB then
                local sA, onA = WorldToScreen(pA.Position)
                local sB, onB = WorldToScreen(pB.Position)
                if onA and onB then
                    line.From=sA; line.To=sB
                    line.Color=color; line.Transparency=alpha
                    line.Thickness=1; line.Visible=true
                else line.Visible=false end
            else line.Visible=false end
        end
    else
        HideLines(data.Skeleton)
    end

    -- Name
    if Settings.NameESP then
        local box = GetBoundingBox(char)
        if box then
            data.NameLabel.Text         = player.Name
            data.NameLabel.Size         = Settings.ESPTextSize
            data.NameLabel.Color        = color
            data.NameLabel.Transparency = alpha
            data.NameLabel.Position     = Vector2.new(box.Center.X, box.Y - Settings.ESPTextSize - 2)
            data.NameLabel.Visible      = true
        end
    else data.NameLabel.Visible=false end

    -- Health Bar
    if Settings.HealthESP then
        local box = GetBoundingBox(char)
        if box then
            local pct = math.clamp(hum.Health/hum.MaxHealth, 0, 1)
            local bx  = box.X - 6
            local byB = box.Y + box.Height
            local byT = box.Y
            local fillY = byB - (box.Height * pct)
            local hcol = Color3.fromRGB(math.floor(255*(1-pct)), math.floor(255*pct), 0)
            data.HealthBG.From=Vector2.new(bx,byT); data.HealthBG.To=Vector2.new(bx,byB)
            data.HealthBG.Thickness=Settings.ESPOutlineThick+3; data.HealthBG.Transparency=alpha; data.HealthBG.Visible=true
            data.HealthFill.From=Vector2.new(bx,fillY); data.HealthFill.To=Vector2.new(bx,byB)
            data.HealthFill.Color=hcol; data.HealthFill.Thickness=Settings.ESPOutlineThick+1
            data.HealthFill.Transparency=alpha; data.HealthFill.Visible=true
        end
    else data.HealthBG.Visible=false; data.HealthFill.Visible=false end

    -- Distance
    if Settings.DistanceESP then
        local box = GetBoundingBox(char)
        if box then
            data.DistLabel.Text=string.format("%.0fm", dist)
            data.DistLabel.Size=math.max(10,Settings.ESPTextSize-2)
            data.DistLabel.Transparency=alpha
            data.DistLabel.Position=Vector2.new(box.Center.X, box.Y+box.Height+2)
            data.DistLabel.Visible=true
        end
    else data.DistLabel.Visible=false end

    -- Tracer
    if Settings.TracerESP then
        local rootScreen, rootOn = WorldToScreen(root.Position)
        if rootOn then
            local center = GetCenter()
            data.Tracer.From=Vector2.new(center.X, Camera.ViewportSize.Y)
            data.Tracer.To=rootScreen; data.Tracer.Color=color
            data.Tracer.Transparency=alpha; data.Tracer.Thickness=1; data.Tracer.Visible=true
        else data.Tracer.Visible=false end
    else data.Tracer.Visible=false end

    -- Offscreen Arrow
    if Settings.OffscreenArrow then
        local rootScreen, rootOn = WorldToScreen(root.Position)
        if not rootOn then
            local center    = GetCenter()
            local direction = (rootScreen - center).Unit
            local angle     = math.atan2(direction.Y, direction.X)
            local margin    = 60
            local vp        = Camera.ViewportSize
            local arrowPos  = Vector2.new(
                math.clamp(center.X + direction.X * (vp.X/2 - margin), margin, vp.X-margin),
                math.clamp(center.Y + direction.Y * (vp.Y/2 - margin), margin, vp.Y-margin)
            )
            local size = 10
            local tip  = arrowPos + direction * size
            local perp = Vector2.new(-direction.Y, direction.X) * (size*0.5)
            data.Arrow.PointA   = tip
            data.Arrow.PointB   = arrowPos + perp
            data.Arrow.PointC   = arrowPos - perp
            data.Arrow.Color    = color
            data.Arrow.Transparency = alpha
            data.Arrow.Filled   = true
            data.Arrow.Visible  = true
        else data.Arrow.Visible=false end
    else data.Arrow.Visible=false end

    -- Weapon ESP
    if Settings.WeaponESP then
        local box = GetBoundingBox(char)
        if box then
            local tool = char:FindFirstChildOfClass("Tool")
            local weaponName = tool and tool.Name or ""
            if weaponName ~= "" then
                data.WeaponLabel.Text=weaponName
                data.WeaponLabel.Size=math.max(10,Settings.ESPTextSize-2)
                data.WeaponLabel.Transparency=alpha
                data.WeaponLabel.Position=Vector2.new(box.Center.X, box.Y+box.Height+Settings.ESPTextSize+4)
                data.WeaponLabel.Visible=true
            else data.WeaponLabel.Visible=false end
        end
    else data.WeaponLabel.Visible=false end

    -- Highlight (Chams)
    if Settings.HighlightESP then
        data.Highlight.FillColor=color
        data.Highlight.OutlineColor=color
        data.Highlight.FillTransparency=math.clamp(Settings.ESPTransparency+0.5,0,1)
        data.Highlight.OutlineTransparency=0
        data.Highlight.Enabled=true
    else data.Highlight.Enabled=false end
end

local function UpdateRadar()
    if not Settings.RadarESP then DestroyRadar(); return end
    if not RadarGui then CreateRadar() end
    if not RadarFrame then return end

    local localChar = LocalPlayer.Character
    local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")
    if not localRoot then return end

    local size  = Settings.RadarSize
    local range = Settings.RadarRange
    local camCF = Camera.CFrame

    -- Track existing dots
    local seen = {}

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if not IsAlive(player) then
            if RadarDots[player] then
                RadarDots[player]:Destroy()
                RadarDots[player] = nil
            end
            continue
        end
        local char = player.Character
        if not char then continue end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then continue end

        local relPos = localRoot.CFrame:PointToObjectSpace(root.Position)
        local rx = math.clamp((relPos.X / range) * 0.5 + 0.5, 0.05, 0.95)
        local ry = math.clamp((relPos.Z / range) * 0.5 + 0.5, 0.05, 0.95)

        if not RadarDots[player] then
            local dot = Instance.new("Frame")
            dot.Size = UDim2.new(0,6,0,6)
            dot.AnchorPoint = Vector2.new(0.5,0.5)
            dot.BackgroundColor3 = GetESPColor(player)
            dot.BorderSizePixel = 0
            dot.ZIndex = 2
            dot.Parent = RadarFrame
            local dc = Instance.new("UICorner")
            dc.CornerRadius = UDim.new(1,0)
            dc.Parent = dot

            local nameTag = Instance.new("TextLabel")
            nameTag.Size = UDim2.new(0,60,0,12)
            nameTag.Position = UDim2.new(0.5,-30,0,-14)
            nameTag.BackgroundTransparency = 1
            nameTag.TextColor3 = Color3.fromRGB(255,255,255)
            nameTag.TextSize = 9
            nameTag.Font = Enum.Font.GothamBold
            nameTag.Text = player.Name
            nameTag.Parent = dot

            RadarDots[player] = dot
        end

        RadarDots[player].Position = UDim2.new(rx, 0, ry, 0)
        RadarDots[player].BackgroundColor3 = GetESPColor(player)
        seen[player] = true
    end

    -- Remove stale dots
    for player, dot in pairs(RadarDots) do
        if not seen[player] then
            dot:Destroy()
            RadarDots[player] = nil
        end
    end
end

    HeatmapDrawLines = {}
    if not Settings.Heatmap then return end
    local maxVal = 1
    for _, v in pairs(HeatmapData) do
        if v > maxVal then maxVal = v end
    end
    for key, val in pairs(HeatmapData) do
        local parts = string.split(key, "_")
        local wx = tonumber(parts[1]) or 0
        local wz = tonumber(parts[2]) or 0
        local worldPos = Vector3.new(wx, 0, wz)
        local screen, on = WorldToScreen(worldPos)
        if on then
            local t = val / maxVal
            local dot = D("Circle", {
                Position=screen,
                Radius=math.clamp(t*12, 3, 15),
                Color=Color3.fromRGB(math.floor(255*t), math.floor(255*(1-t)), 0),
                Filled=true,
                Transparency=0.5,
                Visible=true,
            })
            table.insert(HeatmapDrawLines, dot)
        end
    end
end

-- ══════════════════════════════════════════
-- AIMBOT TARGET SELECTION
-- ══════════════════════════════════════════
local LockedTarget = nil

local function GetAimKeyPressed()
    local kc = Settings.AimKeyCode
    if kc == "MouseButton2" then
        return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
    elseif kc == "MouseButton1" then
        return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    else
        local ok, key = pcall(function() return Enum.KeyCode[kc] end)
        if ok then return UserInputService:IsKeyDown(key) end
    end
    return false
end

local function GetTarget()
    -- if locked, verify still alive
    if Settings.TargetLock and LockedTarget then
        if IsAlive(LockedTarget) and not IsTeammate(LockedTarget) then
            local char = LockedTarget.Character
            if char then
                local part = char:FindFirstChild(Settings.TargetPart)
                if part then return LockedTarget end
            end
        end
        LockedTarget = nil
    end

    local camPos    = Camera.CFrame.Position
    local best      = nil
    local bestScore = math.huge

    local dynFOV = Settings.AimFOV
    if Settings.FOVDynamic then
        -- scale FOV inversely with distance
        local closestDist = math.huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p==LocalPlayer then continue end
            if not IsAlive(p) then continue end
            if IsTeammate(p) then continue end
            local root = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local d = (root.Position-camPos).Magnitude
                if d < closestDist then closestDist=d end
            end
        end
        if closestDist < math.huge then
            dynFOV = math.clamp(Settings.AimFOV * (1 + (500-math.min(closestDist,500))/500), Settings.AimFOV, Settings.AimFOV*2)
        end
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player==LocalPlayer then continue end
        if not IsAlive(player) then continue end
        if IsTeammate(player) then continue end
        local char = player.Character
        if not char then continue end
        local part = char:FindFirstChild(Settings.TargetPart)
        if not part then continue end
        local localChar = LocalPlayer.Character
        local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")
        if Settings.WallCheck and localRoot then
            if HasWall(localRoot.Position, part.Position) then continue end
        end

        local score
        if Settings.TargetPriority == "FOV" then
            score = GetFOVDist(part.Position)
        elseif Settings.TargetPriority == "Distance" then
            score = (part.Position - camPos).Magnitude
        elseif Settings.TargetPriority == "Health" then
            local hum = char:FindFirstChildOfClass("Humanoid")
            score = hum and hum.Health or math.huge
        else
            score = GetFOVDist(part.Position)
        end

        -- FOV gate (only for FOV mode)
        if Settings.TargetPriority == "FOV" then
            if score > dynFOV then continue end
        else
            -- still require within FOV circle
            if GetFOVDist(part.Position) > dynFOV then continue end
        end

        if score < bestScore then
            bestScore = score
            best = player
        end
    end

    if best and Settings.TargetLock then LockedTarget = best end
    return best
end

-- GHOST REPLAY TIMER
-- ══════════════════════════════════════════
local ghostTimer = 0

-- ══════════════════════════════════════════
-- HEATMAP TIMER
-- ══════════════════════════════════════════
local heatTimer = 0

-- ══════════════════════════════════════════
-- RESOURCE COUNTER TIMER
-- ══════════════════════════════════════════
local resourceTimer = 0

-- ══════════════════════════════════════════
-- MAIN RENDER LOOP
-- ══════════════════════════════════════════
Track(RunService.RenderStepped:Connect(function(dt)
    -- FOV Circle
    local dynFOVDisplay = Settings.AimFOV
    fovCircle.Position    = GetCenter()
    fovCircle.Radius      = dynFOVDisplay
    fovCircle.Transparency = Settings.FOVTransparency
    fovCircle.Visible     = Settings.Aimbot and Settings.FOVVisible

    -- ESP
    if Settings.ESP then
        for _, player in ipairs(Players:GetPlayers()) do
            if player == LocalPlayer then continue end
            if not ESPObjects[player] then CreateESP(player) end
            if ESPObjects[player] then UpdateESP(player, ESPObjects[player]) end
        end
        for player in pairs(ESPObjects) do
            if not Players:FindFirstChild(player.Name) then
                RemoveESP(player)
            end
        end
    else
        for player in pairs(ESPObjects) do RemoveESP(player) end
    end

    -- Radar
    UpdateRadar()

    -- Aimbot
    if Settings.Aimbot and GetAimKeyPressed() then
        local target = GetTarget()
        if target and target.Character then
            local part = target.Character:FindFirstChild(Settings.TargetPart)
            if part then
                local aimPos = part.Position
                -- Aim prediction
                if Settings.AimPrediction then
                    local root = target.Character:FindFirstChild("HumanoidRootPart")
                    if root then
                        local vel = root.AssemblyLinearVelocity
                        aimPos = aimPos + vel * Settings.PredictStrength
                    end
                end
                local _, onScreen = WorldToScreen(aimPos)
                if onScreen then
                    local currentCF = Camera.CFrame
                    local targetCF  = CFrame.lookAt(currentCF.Position, aimPos)
                    Camera.CFrame   = currentCF:Lerp(targetCF, Settings.AimSmoothing)
                end
            end
        end
    end

    -- Ghost replay timers
    ghostTimer = ghostTimer + dt
    if ghostTimer >= 0.1 then
        ghostTimer = 0
        RecordGhostPoint()
        DrawGhostReplay()
    end

    heatTimer = heatTimer + dt
    if heatTimer >= 0.5 then
        heatTimer = 0
        RecordHeatmap()
        DrawHeatmap()
    end

    resourceTimer = resourceTimer + dt
    if resourceTimer >= 2 then
        resourceTimer = 0
        UpdateResourceCounter()
    end
end))

-- Player leaving
Track(Players.PlayerRemoving:Connect(function(player)
    if ESPObjects[player] then RemoveESP(player) end
    if RadarDots[player] then RadarDots[player]:Destroy(); RadarDots[player]=nil end
end))
